// 1. Добавление аутентификации через JWT
function TWebModuleMain.CheckAuth(Request: TWebRequest): Boolean;
var
  Token: string;
begin
  Token := Request.CookieFields.Values['auth_token'];
  Result := ValidateJWT(Token);
end;

// 3. Генерация CSRF-токена
function GenerateCSRFToken: string;
begin
  Result := TEncoding.UTF8.GetString(TEncoding.UTF8.GetBytes(GUIDToString(TGUID.NewGuid)));
end;
